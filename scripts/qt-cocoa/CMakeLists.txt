project(CocoaAccessibilityFix VERSION 6.11.2 LANGUAGES C CXX OBJC OBJCXX)
set(QT_SKIP_AUTO_PLUGIN_INCLUSION ON)
set(QT_GENERATE_SBOM OFF)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS BuildInternals Core CorePrivate Gui GuiPrivate Widgets)
qt_build_repo_begin()
foreach(fw Foundation AppKit Carbon CoreServices CoreVideo IOKit IOSurface Metal QuartzCore UniformTypeIdentifiers)

endforeach()
add_subdirectory("${QTBASE_SOURCE_DIR}/src/plugins/platforms/cocoa" cocoa)
add_executable(ownership_test ownership_test.mm)
target_link_libraries(ownership_test PRIVATE Qt6::Widgets ${FWAppKit})
